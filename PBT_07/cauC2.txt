/*
==================================================
BÀI TẬP: TÍNH HÓA ĐƠN NHÀ HÀNG
==================================================

Quy tắc:

- Tổng > 500.000đ  => giảm 10%
- Tổng > 1.000.000đ => giảm 15%
- Thứ 3 (Wednesday) => giảm thêm 5%
- VAT 8%
- Tip 5% (tùy chọn)

==================================================
*/

const monAn = [
    {
        ten: "Phở bò",
        gia: 65000,
        soLuong: 2
    },
    {
        ten: "Trà đá",
        gia: 5000,
        soLuong: 3
    },
    {
        ten: "Bún chả",
        gia: 55000,
        soLuong: 1
    }
];

const ngay = "Wednesday";
const coTip = true;

/*
==================================================
TÍNH TỔNG TIỀN
==================================================
*/

let tongTien = 0;

for (let i = 0; i < monAn.length; i++) {
    tongTien += monAn[i].gia * monAn[i].soLuong;
}

/*
==================================================
TÍNH GIẢM GIÁ
==================================================
*/

let phanTramGiam = 0;

if (tongTien > 1000000) {
    phanTramGiam = 15;
}
else if (tongTien > 500000) {
    phanTramGiam = 10;
}

if (ngay === "Wednesday") {
    phanTramGiam += 5;
}

const tienGiam = tongTien * phanTramGiam / 100;

const tienSauGiam = tongTien - tienGiam;

/*
==================================================
VAT
==================================================
*/

const VAT = tienSauGiam * 0.08;

/*
==================================================
TIP
==================================================
*/

let tip = 0;

if (coTip) {
    tip = tienSauGiam * 0.05;
}

/*
==================================================
THANH TOÁN CUỐI CÙNG
==================================================
*/

const thanhToan = tienSauGiam + VAT + tip;

/*
==================================================
IN HÓA ĐƠN
==================================================
*/

console.log("╔══════════════════════════════════════════════╗");
console.log("║              HÓA ĐƠN NHÀ HÀNG              ║");
console.log("╠══════════════════════════════════════════════╣");

for (let i = 0; i < monAn.length; i++) {

    const thanhTien =
        monAn[i].gia * monAn[i].soLuong;

    console.log(
        `${i + 1}. ${monAn[i].ten} x${monAn[i].soLuong} @${monAn[i].gia.toLocaleString()}đ = ${thanhTien.toLocaleString()}đ`
    );
}

console.log("╠══════════════════════════════════════════════╣");

console.log(
    "Tổng cộng: ".padEnd(25) +
    tongTien.toLocaleString() + "đ"
);

console.log(
    `Giảm giá (${phanTramGiam}%): `.padEnd(25) +
    tienGiam.toLocaleString() + "đ"
);

console.log(
    "VAT (8%): ".padEnd(25) +
VAT.toLocaleString() + "đ"
);

console.log(
    "Tip (5%): ".padEnd(25) +
    tip.toLocaleString() + "đ"
);

console.log("╠══════════════════════════════════════════════╣");

console.log(
    "THANH TOÁN: ".padEnd(25) +
    thanhToan.toLocaleString() + "đ"
);

console.log("╚══════════════════════════════════════════════╝");
